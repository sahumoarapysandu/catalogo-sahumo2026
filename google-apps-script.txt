CODIGO PARA GOOGLE APPS SCRIPT — STOCK, PEDIDOS Y VENTAS DE FERIA
=========================================================================
Este es el código que tenés que pegar en Google Apps Script. Reemplaza
al que tenías antes: ahora recibe los datos como formulario en vez de
JSON, que es más confiable y evita que algunos navegadores pierdan el
pedido en el camino.

Instrucciones completas en LEEME.txt.

La hoja de STOCK debe tener 7 columnas: id | nombre | categoria | marca | aroma | stock | ultima_venta
(esta la tenés que crear vos, como siempre)

La hoja de PEDIDOS se crea SOLA la primera vez que llega un pedido
del catálogo online, con estas columnas: fecha | nombre | telefono | pedido | total

La hoja de VENTASFERIA se crea SOLA la primera vez que un vendedor
carga una venta desde panel-ventas.html, con estas columnas:
lugar | categoria | producto | cantidad | precio | hora | dia

--------------------------------------------------------------------

// ID de la hoja de Google Sheets a la que se conecta este script.
// Sacado de: https://docs.google.com/spreadsheets/d/1-DhFJP2rol-babmzhORAjQan_jIs08XM8B9XRX3uvow/edit
var SPREADSHEET_ID = "1-DhFJP2rol-babmzhORAjQan_jIs08XM8B9XRX3uvow";

function doGet(e) {
  var sheet = SpreadsheetApp.openById(SPREADSHEET_ID).getActiveSheet();
  var data = sheet.getDataRange().getValues();
  var rows = [];
  for (var i = 1; i < data.length; i++) {
    if (data[i][0] === "") continue;
    rows.push({
      id: data[i][0],
      nombre: data[i][1],
      categoria: data[i][2] || "",
      marca: data[i][3] || "",
      aroma: data[i][4] || "",
      stock: data[i][5],
      ultimaVenta: data[i][6] || ""
    });
  }
  return ContentService
    .createTextOutput(JSON.stringify(rows))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var ss = SpreadsheetApp.openById(SPREADSHEET_ID);
  var sheet = ss.getActiveSheet();
  var p = e.parameter;
  var action = p.action;

  function findRow(values, id) {
    for (var i = 1; i < values.length; i++) {
      if (String(values[i][0]) === String(id)) return i + 1;
    }
    return -1;
  }

  if (action === "set") {
    var values = sheet.getDataRange().getValues();
    var row = findRow(values, p.id);
    if (row === -1) {
      sheet.appendRow([p.id, p.nombre, p.categoria || "", p.marca || "", p.aroma || "", p.stock, ""]);
    } else {
      sheet.getRange(row, 2).setValue(p.nombre);
      sheet.getRange(row, 3).setValue(p.categoria || "");
      sheet.getRange(row, 4).setValue(p.marca || "");
      sheet.getRange(row, 5).setValue(p.aroma || "");
      sheet.getRange(row, 6).setValue(p.stock);
    }
  }

  if (action === "decrement") {
    var values = sheet.getDataRange().getValues();
    var items = JSON.parse(p.items || "[]");
    var fecha = Utilities.formatDate(new Date(), "GMT-3", "dd/MM/yyyy HH:mm");
    items.forEach(function(item) {
      var row = findRow(values, item.id);
      if (row === -1) {
        sheet.appendRow([item.id, "", "", "", "", 0, fecha]);
      } else {
        var current = sheet.getRange(row, 6).getValue();
        var nuevo = Math.max(0, current - item.cantidad);
        sheet.getRange(row, 6).setValue(nuevo);
        sheet.getRange(row, 7).setValue(fecha);
      }
    });
  }

  if (action === "pedido") {
    var pedidosSheet = ss.getSheetByName("Pedidos");
    if (!pedidosSheet) {
      pedidosSheet = ss.insertSheet("Pedidos");
      pedidosSheet.appendRow(["fecha", "nombre", "telefono", "pedido", "total"]);
    }
    var fecha = Utilities.formatDate(new Date(), "GMT-3", "dd/MM/yyyy HH:mm");
    pedidosSheet.appendRow([
      fecha,
      p.nombre || "",
      p.telefono || "",
      p.pedido || "",
      p.total || 0
    ]);
  }

  if (action === "ventaferia") {
    var ventasSheet = ss.getSheetByName("VentasFeria");
    if (!ventasSheet) {
      ventasSheet = ss.insertSheet("VentasFeria");
      ventasSheet.appendRow(["lugar", "categoria", "producto", "cantidad", "precio", "hora", "dia"]);
    }
    ventasSheet.appendRow([
      p.lugar || "",
      p.categoria || "",
      p.producto || "",
      p.cantidad || "",
      p.precio || "",
      p.hora || "",
      p.dia || ""
    ]);
  }

  return ContentService
    .createTextOutput(JSON.stringify({ status: "ok" }))
    .setMimeType(ContentService.MimeType.JSON);
}
